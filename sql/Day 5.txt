1 Write a query to list employee names who work in department 'D-104' or 'D-102', using a subquery
with IN.
 SELECT EMPNAME
 FROM EMPLOYEE
 WHERE DEPTID IN ( SELECT DEPTID 
                                  FROM DEPT
                                  WHERE DEPTID IN (40, 20)); 

2 Write a query to list employee names working in the 'Development' department.
 SELECT EMPNAME
 FROM EMPLOYEE
 WHERE DEPTID IN ( SELECT DEPTID 
				  FROM DEPT
                                  WHERE DEPTNAME = 'RESEARCH');

3 Write a query to count employees in the 'Research' department whose salary is less than at least
one salary in department 30 (hint: use SAL < ANY).
 SELECT COUNT(*)
 FROM EMPLOYEE
 WHERE DEPTID IN ( SELECT DEPTID
                                  FROM DEPT
                                  WHERE DEPTNAME = 'RESEARCH')  AND SAL < ANY ( SELECT SAL
                                                                                                                          FROM EMPLOYEE
															  WHERE DEPTID = 30);

4 Write a query to find employees earning more than any employee whose job is 'ANALYST' (use >
ANY).
 SELECT EMPNAME
 FROM EMPLOYEE
 WHERE SAL > ANY  ( SELECT SALARY
				    FROM EMPLOYEE
				    WHERE JOB = 'ANALYST');
